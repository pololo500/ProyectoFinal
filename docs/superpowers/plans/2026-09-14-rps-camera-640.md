```python
class _FakeCapture:
    def __init__(self, honor_set: bool = True, actual_w: float = 320.0, actual_h: float = 240.0) -> None:
        self.honor_set = honor_set
        self._props: dict[object, float] = {}
        self.actual_w = actual_w
        self.actual_h = actual_h

    def set(self, prop: object, value: float) -> bool:
        self._props[prop] = float(value)
        return True

    def get(self, prop: object) -> float:
        import cv2

        if self.honor_set:
            return self._props.get(prop, 0.0)
        if prop == cv2.CAP_PROP_FRAME_WIDTH:
            return self.actual_w
        if prop == cv2.CAP_PROP_FRAME_HEIGHT:
            return self.actual_h
        return 0.0
```

```python
def apply_camera_capture_size(capture: Any) -> tuple[int, int]:
    """Pide 640×480. Devuelve el tamaño que reporta el driver (puede ignorar el set)."""
    capture.set(cv2.CAP_PROP_FRAME_WIDTH, CAMERA_CAPTURE_WIDTH)
    capture.set(cv2.CAP_PROP_FRAME_HEIGHT, CAMERA_CAPTURE_HEIGHT)
    actual_w = int(capture.get(cv2.CAP_PROP_FRAME_WIDTH) or 0)
    actual_h = int(capture.get(cv2.CAP_PROP_FRAME_HEIGHT) or 0)
    return actual_w, actual_h
```
